# Description: Create and configure the required database tables in Supabase according to the schema requirements.
# Details:
1. Create the following tables in Supabase:

   a. users
   ```sql
   CREATE TABLE users (
     id UUID REFERENCES auth.users NOT NULL PRIMARY KEY,
     name TEXT,
     email TEXT UNIQUE NOT NULL,
     profile_image_url TEXT,
     certification_type TEXT,
     target_exam_date DATE,
     streak_count INTEGER DEFAULT 0,
     last_streak_date DATE,
     created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
     updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
   );
   ```

   b. questions
   ```sql
   CREATE TABLE questions (
     id UUID DEFAULT uuid_generate_v4() PRIMARY KEY,
     content TEXT NOT NULL,
     options JSONB NOT NULL,
     correct_answer TEXT NOT NULL,
     explanation TEXT NOT NULL,
     certification_type TEXT NOT NULL,
     category TEXT NOT NULL,
     difficulty INTEGER NOT NULL,
     tags TEXT[],
     created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
   );
   ```

   c. diagnosis_results
   ```sql
   CREATE TABLE diagnosis_results (
     id UUID DEFAULT uuid_generate_v4() PRIMARY KEY,
     user_id UUID REFERENCES users(id) ON DELETE CASCADE,
     scores JSONB NOT NULL,
     weak_areas TEXT[],
     created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
   );
   ```

   d. study_records
   ```sql
   CREATE TABLE study_records (
     id UUID DEFAULT uuid_generate_v4() PRIMARY KEY,
     user_id UUID REFERENCES users(id) ON DELETE CASCADE,
     question_id UUID REFERENCES questions(id) ON DELETE CASCADE,
     user_answer TEXT,
     is_correct BOOLEAN NOT NULL,
     time_spent INTEGER,
     created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
   );
   ```

   e. wrong_answers
   ```sql
   CREATE TABLE wrong_answers (
     id UUID DEFAULT uuid_generate_v4() PRIMARY KEY,
     user_id UUID REFERENCES users(id) ON DELETE CASCADE,
     question_id UUID REFERENCES questions(id) ON DELETE CASCADE,
     wrong_count INTEGER DEFAULT 1,
     last_wrong_date TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
     next_review_date TIMESTAMP WITH TIME ZONE,
     graduated BOOLEAN DEFAULT FALSE,
     created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
     updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
   );
   ```

   f. daily_sets
   ```sql
   CREATE TABLE daily_sets (
     id UUID DEFAULT uuid_generate_v4() PRIMARY KEY,
     user_id UUID REFERENCES users(id) ON DELETE CASCADE,
     question_ids UUID[] NOT NULL,
     completed BOOLEAN DEFAULT FALSE,
     completed_at TIMESTAMP WITH TIME ZONE,
     created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
   );
   ```

2. Set up Row Level Security (RLS) policies for each table
3. Create database triggers for updating timestamps
4. Create database functions for common operations
5. Set up database indexes for performance optimization
